## 🗄️ SUPABASE DATABASE SCHEMA — run all of this in Supabase SQL editor

```sql
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- USERS
create table users (
  id uuid primary key default uuid_generate_v4(),
  email text unique not null,
  password_hash text not null,
  full_name text,
  phone text,
  role text default 'customer' check (role in ('customer', 'admin')),
  created_at timestamptz default now()
);

-- PRODUCTS
create table products (
  id uuid primary key default uuid_generate_v4(),
  title text not null,
  handle text unique not null,
  body_html text,
  vendor text,
  product_type text,
  tags text[],
  status text default 'active' check (status in ('active', 'draft', 'archived')),
  images jsonb default '[]',
  variants jsonb default '[]',
  buying_price decimal(10,2),
  selling_price decimal(10,2) not null,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- ORDERS
create table orders (
  id uuid primary key default uuid_generate_v4(),
  order_number serial unique,
  user_id uuid references users(id) on delete set null,
  customer_name text not null,
  customer_email text,
  customer_phone text not null,
  delivery_name text,
  delivery_phone text,
  delivery_matatu_route text,
  delivery_notes text,
  items jsonb not null,
  subtotal decimal(10,2) not null,
  delivery_fee decimal(10,2) not null default 200,
  total decimal(10,2) not null,
  payment_method text check (payment_method in ('mpesa_till','mpesa_stk','paystack_card','paystack_mobile')),
  payment_status text default 'pending' check (payment_status in ('pending','paid','failed')),
  mpesa_code text,
  paystack_reference text,
  order_status text default 'pending' check (order_status in ('pending','confirmed','processing','shipped','delivered','cancelled')),
  admin_notes text,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- FAVOURITES
create table favourites (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid references users(id) on delete cascade,
  product_id uuid references products(id) on delete cascade,
  created_at timestamptz default now(),
  unique(user_id, product_id)
);

-- BROWSING HISTORY
create table browsing_history (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid references users(id) on delete set null,
  session_id text,
  product_id uuid references products(id) on delete cascade,
  viewed_at timestamptz default now()
);

-- SETTINGS
create table settings (
  key text primary key,
  value text,
  updated_at timestamptz default now()
);

-- Seed default settings
insert into settings (key, value) values
  ('whatsapp_number', '254922748842'),

  ('shop_name', 'Antar Scents'),
  ('tagline', 'Discover Your Signature Scent'),
  ('announcement_bar', '🚚 Free delivery on orders above KES 5,000 | 📞 +254922748842'),
  ('mpesa_till_number', '000000'),
  ('delivery_fee_cbd', '200'),
  ('delivery_fee_nairobi', '300'),
  ('delivery_fee_outside', '500'),
  ('delivery_fee_far', '1000');

-- Row Level Security
alter table users enable row level security;
alter table products enable row level security;
alter table orders enable row level security;
alter table favourites enable row level security;
alter table browsing_history enable row level security;
alter table settings enable row level security;

-- Products: anyone can read active products
create policy "Public read active products" on products for select using (status = 'active');

-- Settings: anyone can read
create policy "Public read settings" on settings for select using (true);

-- Orders: users see own orders
create policy "Users see own orders" on orders for select using (auth.uid() = user_id);

-- Favourites: users manage their own
create policy "Users manage own favourites" on favourites for all using (auth.uid() = user_id);

## 🔧 BACKEND — EXPRESS API (All Routes)

Base URL: `https://your-render-app.onrender.com`

All responses follow: `{ success: true|false, data: {...}, message: "..." }`

Use Zod for all input validation. Add express-rate-limit on payment routes. Add helmet, cors, morgan middleware.

### `GET /health` — health check, returns `{ status: "ok" }`

### Auth Routes `/api/auth`
```
POST /api/auth/register        — { full_name, email, phone, password } → creates user, returns JWT
POST /api/auth/login           — { email, password } → returns JWT + user object
GET  /api/auth/me              — protected, returns current user from JWT
POST /api/auth/change-password — protected, { current_password, new_password }
```

### Product Routes `/api/products`
```
GET    /api/products              — public, query params: ?category=&vendor=&tags=&min_price=&max_price=&sort=&page=&limit=&search=
GET    /api/products/:slug        — public, returns full product
POST   /api/products              — admin only, create product
PUT    /api/products/:id          — admin only, update product (including buying_price, selling_price)
DELETE /api/products/:id          — admin only
POST   /api/products/import-csv   — admin only, multipart form with CSV file
```

**CSV Import:** Accept standard Shopify product export CSV. Map these columns:
- Title → title
- Handle → handle
- Body (HTML) → body_html
- Vendor → vendor
- Type → product_type
- Tags → tags (split by comma)
- Variant Price → selling_price (first variant)
- Variant SKU → variants[0].sku
- Image Src → images[0].src
- Status → status

Parse with `csv-parse`, bulk insert via Supabase, return count of imported products.

### Order Routes `/api/orders`
```
POST /api/orders                        — create order (guest or authenticated)
GET  /api/orders                        — admin: all orders with filters; customer: own orders
GET  /api/orders/:id                    — order detail (admin or owner)
PUT  /api/orders/:id/status             — admin only, update order_status
PUT  /api/orders/:id/payment-status     — admin only, manually mark as paid
POST /api/orders/:id/mpesa-code         — customer submits M-Pesa confirmation code
GET  /api/orders/track                  — public, query: ?order_number=&phone= returns status
```

### Payment Routes `/api/payments`

**M-Pesa (Daraja API):**
```
POST /api/payments/mpesa/stk-push
  Body: { phone, amount, order_id }
  → Get OAuth token from Daraja
  → Generate timestamp + password (base64 of shortcode+passkey+timestamp)
  → POST to https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest (use production URL in prod)
  → Returns CheckoutRequestID, save to order

POST /api/payments/mpesa/callback
  → Daraja sends result here
  → If ResultCode = 0: update order payment_status = 'paid', save MpesaReceiptNumber as mpesa_code
  → If ResultCode != 0: update payment_status = 'failed'
```

**Paystack:**
```
POST /api/payments/paystack/initialize
  Body: { email, amount, order_id }
  → POST to https://api.paystack.co/transaction/initialize with secret key
  → Return authorization_url and reference

GET /api/payments/paystack/verify/:reference
  → GET https://api.paystack.co/transaction/verify/:reference
  → If status = 'success': update order payment_status = 'paid'
  → Return order updated status
```

### Settings Routes `/api/settings`
```
GET /api/settings        — public, returns all key-value pairs as object
PUT /api/settings/:key   — admin only, update a setting value
```

### Recommendations Route `/api/recommendations`
```
GET /api/recommendations?product_id=&user_id=&session_id=&limit=8
  → Fetch current product's tags, vendor, product_type
  → Query products with overlapping tags OR same vendor OR same product_type
  → If user_id provided: also pull their browsing_history last 20 products, find common tags, weight those products higher
  → Exclude the current product
  → Return sorted by relevance score, limit 8
```

### Browsing History `/api/browsing`
```
POST /api/browsing     — { product_id, user_id (optional), session_id } → upsert to browsing_history
```

### Admin Finance Routes `/api/admin`
```
GET /api/admin/finances
  Query params: ?from=YYYY-MM-DD&to=YYYY-MM-DD
  → Join orders (paid only) with items, calculate:
     revenue = sum of order totals
     cost = sum of (buying_price × qty) per line item
     profit = revenue - cost
     profit_margin = (profit / revenue) × 100
  → Group by day/month
  → Return: { summary: { total_revenue, total_cost, net_profit, profit_margin, total_orders, avg_order_value }, chart_data: [...], orders_breakdown: [...] }

GET /api/admin/finances/export
  → Same as above but returns CSV file download
  → Columns: Order #, Date, Customer, Items, Revenue (KES), Cost (KES), Profit (KES)

GET /api/admin/customers
  → List all users with role='customer'
  → Include: order_count, total_spent (sum of order totals)

GET /api/admin/dashboard
  → Returns: today_sales, this_month_revenue, total_orders, pending_orders_count, total_products, low_stock_products (variants with inventory < 5), recent_orders (last 10), top_products (by order frequency, last 30 days), revenue_chart (last 30 days by day)
```

## 🔒 AUTH & MIDDLEWARE

**Backend middleware:**
```javascript
// authMiddleware.js — verify JWT
const authMiddleware = (req, res, next) => {
  const token = req.headers.authorization?.split(' ')[1];
  if (!token) return res.status(401).json({ success: false, message: 'No token' });
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;
    next();
  } catch {
    return res.status(401).json({ success: false, message: 'Invalid token' });
  }
};

// adminMiddleware.js — check role
const adminMiddleware = (req, res, next) => {
  if (req.user.role !== 'admin') {
    return res.status(403).json({ success: false, message: 'Admin only' });
  }
  next();
};
```

**Frontend auth:**
- Store JWT in localStorage
- Create `useAuth()` hook: `{ user, login, logout, isAdmin, loading }`
- Wrap admin pages with `<AdminGuard>` component that redirects to `/account/login` if not admin
- Wrap account pages with `<AuthGuard>` component

## 🧠 RECOMMENDATION ENGINE LOGIC

```javascript
// GET /api/recommendations?product_id=X&user_id=Y&limit=8
async function getRecommendations(product_id, user_id, limit = 8) {
  // 1. Get current product details
  const product = await supabase.from('products').select('*').eq('id', product_id).single();
  const { tags, vendor, product_type } = product.data;

  // 2. Score all other active products
  const { data: allProducts } = await supabase.from('products')
    .select('*').eq('status', 'active').neq('id', product_id);

  const scored = allProducts.map(p => {
    let score = 0;
    if (p.vendor === vendor) score += 3;
    if (p.product_type === product_type) score += 2;
    const sharedTags = (p.tags || []).filter(t => (tags || []).includes(t));
    score += sharedTags.length * 1.5;
    return { ...p, score };
  }).sort((a, b) => b.score - a.score);

  // 3. If user_id: boost products matching user's browsing history tags
  if (user_id) {
    const { data: history } = await supabase.from('browsing_history')
      .select('product_id').eq('user_id', user_id).order('viewed_at', { ascending: false }).limit(20);
    // additional weighting logic here
  }

  return scored.slice(0, limit);
}
